+/** ページの中の商品の行（重複を除く）。collector/myca.py の parse_rows と同じ。 */
+export function parseRows(page) {
+  const t = page.replaceAll('\\"', '"');
+  const out = [];
+  const seen = new Set();
+  for (const m of t.matchAll(ROW)) {
+    const id = Number(m[1]);
+    if (seen.has(id)) continue;
+    seen.add(id);
+    out.push({
+      item_id: id, name: JSON.parse(`"${m[2]}"`), rarity: val(m[3]) || null, expansion: val(m[4]),
+      cardnumber: val(m[5]), price: m[6] === "null" ? null : Number(m[6]), image_url: val(m[7]), category: m[8],
+    });
+  }
+  return out;

+/** BOX の一覧の行から、弾の BOX（名前が「… BOX」）を拾う。collector/myca.py の box_sets と同じ。 */
+export function boxSets(rows) {
+  const byName = new Map(rows.map((r) => [r.name, r]));
+  const out = [];
+  const seen = new Set();
+  for (const r of rows) {

+      const r = await fetchPage(`/${GENRE}/list?category=BOX${p > 1 ? `&page=${p}` : ""}`);

+        state.phase = "done";

+      state.setIdx += 1;
+      state.cur = null;
+    };

+        nextSet();
+        continue;
+      }
+      const r = await fetchPage(`/${GENRE}/items/box/${cur.ids[cur.idx]}`);
+      cur.idx += 1;
+      if (r.status !== 200) {
+        c.failed += 1;  // 1つの弾が読めなくても、ほかの弾は読む
+        nextSet();
+        continue;
+      }
+      cur.pack = packOf(r.text);
+      continue;
+    }
+    const p = cur.page;

+    let end = false;
+    if (r.status === 404 && p > 1) {
+      end = true;

+      nextSet();
+      continue;

+      if (!fresh.length) end = true;
+      else {
+        cur.cards.push(...fresh);
+        cur.seen.push(...fresh.map((x) => x.item_id));
+        if (p >= MAX_PAGES) end = true;
+        else cur.page = p + 1;
+      }
+    }
+    if (end) {
+      state.done.push({ ...s, pack: cur.pack, cards: cur.cards });
+      c.sets += 1;
+      c.cards += cur.cards.length;
+      nextSet();
+    }
+  }
+  if (state.phase === "done" && !state.finished_at) state.finished_at = now().toISOString();
+  return state;
+}

+    counts: state.counts, sets: state.done,
