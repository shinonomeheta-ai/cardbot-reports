+import { storeKey } from "./store-identity.mjs";

+export class FavSet extends Set {
+  constructor(names) {
+    super(names || []);
+    this.keys_ = new Set([...this].map((n) => storeKey(n)).filter(Boolean));
+  }
+
+  has(name) {
+    if (super.has(name)) return true;
+    const k = storeKey(name);
+    return !!k && !!this.keys_ && this.keys_.has(k);
+  }
+}
+
+/** `names`（Set）から、`name` と同じ鍵の名前をすべて外した新しい Set。 */
+export function removeSame(names, name) {
+  const k = storeKey(name);
+  return new Set([...names].filter((n) => n !== name && !(k && storeKey(n) === k)));
+}
+
+/** `names`（Set）に、`name` と同じ鍵の名前があるか。 */
+export function hasSame(names, name) {
+  if (names.has(name)) return true;
+  const k = storeKey(name);
+  return !!k && [...names].some((n) => storeKey(n) === k);
+}
